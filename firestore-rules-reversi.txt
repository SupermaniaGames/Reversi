    match /reversiRooms/{code} {
      // any signed-in player (guests included) can read a room, so a joiner can look the code up
      allow read: if request.auth != null;

      // create a new room as host; or re-use a code whose old room is more than 6 hours old
      allow create: if request.auth != null
                    && request.resource.data.host == request.auth.uid
                    && request.resource.data.status == 'lobby';
      allow update: if request.auth != null && (
          // 1. a player already in the room
          request.auth.uid == resource.data.host
          || resource.data.players.size() == 2 && (
               resource.data.players[0].uid == request.auth.uid
               || resource.data.players[1].uid == request.auth.uid)
          // 2. the second player joining a lobby (only the players list may change)
          || (resource.data.status == 'lobby'
              && resource.data.players.size() == 1
              && request.resource.data.players.size() == 2
              && request.resource.data.players[1].uid == request.auth.uid
              && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['players']))
          // 3. an expired room code being re-used by a new host
          || (request.resource.data.host == request.auth.uid
              && resource.data.created < request.time.toMillis() - 21600000)
        );
      allow delete: if false;

      match /chat/{msg} {
        allow read: if request.auth != null;
        allow create: if request.auth != null
                      && request.resource.data.uid == request.auth.uid
                      && request.resource.data.text is string
                      && request.resource.data.text.size() <= 200;
        allow update, delete: if false;
      }
    }
